cmake_minimum_required(VERSION 3.25)
project(knmon_cet_probe LANGUAGES CXX ASM_MASM)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The CET evidence probe requires Windows x64 and MSVC.")
endif()
add_executable(knmon-cet-probe Probe.cpp Returns.asm)
target_compile_features(knmon-cet-probe PRIVATE cxx_std_20)
target_compile_definitions(knmon-cet-probe PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_compile_options(knmon-cet-probe PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/guard:cf;/GS;/sdl>")
target_link_options(knmon-cet-probe PRIVATE /GUARD:CF /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA /CETCOMPAT /INCREMENTAL:NO)
